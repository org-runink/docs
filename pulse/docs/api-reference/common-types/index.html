Common types

Common types

Shared messages and enums from commons.proto, and any declared type no method reaches.

Status messages on a stream

Some StreamChunk messages carry no output: they report that the stream is alive. They have empty text, is_final false, and say so in metadata. The keys are listed under Calling the API.

The judgement types

Judgement and JudgementSet come from the shared package runink.ui.judgement.v1. See The judging ladder.

// PaginationRequest allows clients to paginate through list results.

// StreamChunk is the universal wrapper for server-streaming AI responses.
message StreamChunk {
  string text = 1;
  bool is_final = 2;
  string section = 3;      // semantic section marker (e.g. "positioning", "swot", "prescription")
  float progress = 4;      // 0.0 - 1.0
  map<string, string> metadata = 5;
  // Real, backend-verified actions taken while producing this chunk (e.g.
  // "Scraped live content from acme.com") — not model self-report. Sent as
  // one-off chunks alongside text chunks; a chunk with agent_tasks set may
  // have empty text. Same shape as ModelService.Complete's agent_tasks
  // tag convention, generalized onto the wire instead of text-embedded tags.
  repeated string agent_tasks = 6;
  // The assistant's tools: one tool it
  // ran, or refused, for this turn. Only ModelService.Complete sets it.
  AssistantToolEvent tool_event = 7;
  // An action the assistant proposes. Nothing has run: it runs only when the
  // user taps Confirm (ModelService.ConfirmAssistantAction).
  AssistantProposal proposal = 8;
}

// One tool step of the assistant. The client words it from `tool` and
// `status`; `detail` is the server's refusal or failure reason.
message AssistantToolEvent {
  string tool = 1;   // registry name, e.g. "list_leads"; empty for a general event
  string status = 2; // running | done | refused | failed | blocked (the guard refused the message; detail = reason) | step_limit | deadline
  string detail = 3;
  int32 count = 4;   // rows the read returned; meaningful only when counted
  bool counted = 5;
}

// A proposed action, held server-side until it expires. `args` is shown to
// the user; Confirm runs the server's own copy, so the client cannot alter it.
message AssistantProposal {
  string id = 1;
  string tool = 2;
  map<string, string> args = 3;
  string route = 4; // the page the action changes
  google.protobuf.Timestamp expires_at = 5;
}

// DateRange for filtering by date.
message DateRange {
  google.protobuf.Timestamp start = 1;
  google.protobuf.Timestamp end = 2;
}

// KeyValue for generic key-value pairs.
message KeyValue {
  string key = 1;
  string value = 2;
}

// MetricPoint for time-series data.
message MetricPoint {
  google.protobuf.Timestamp timestamp = 1;
  double value = 2;
  string label = 3;
}

message Runner {
  string name = 1;
  string endpoint = 2;
  string state = 13;
  double compute_units = 4;
  string description = 5;
  string type = 6;
  int32 port = 7;
  int32 cpus = 8;
  string memory = 9;
  string image_repository = 10;
}

message Connection {
  string name = 1;
  string type = 2;
  string environment = 3;
  string runner = 4;
  string id = 5;
  map<string, string> properties = 17;
}

message ActionCard {
  string id = 1;
  string title = 2;
  string type = 3;
  string priority = 4;
  string impact = 5;
  string status = 6;
}
