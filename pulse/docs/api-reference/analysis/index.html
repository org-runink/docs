      Analysis services

      Package pulse.v1. Every method needs a bearer token.

      DiagnosticService

      The Site Audit: the model-written diagnosis and the rule-based relevance audit. See Site Audit.

      MethodRequest → ResponseNotes
      CreateDiagnosticCreateDiagnosticRequest → DiagnosticCreates the record from your business details. No generation yet.
      GetDiagnosticGetDiagnosticRequest → Diagnostic
      ListDiagnosticsListDiagnosticsRequest → ListDiagnosticsResponsePaginated
      GenerateDiagnosticGenerateDiagnosticRequest → stream StreamChunkScrapes the site, streams the analysis, then fills the rest of the app. Fails with FailedPrecondition if the site cannot be read.
      UpdateDiagnosticUpdateDiagnosticRequest → DiagnosticBusiness details and metadata
      ExportDiagnosticPDFExportDiagnosticPDFRequest → ExportDiagnosticPDFResponsePDF bytes and a file name
      AuditURLAuditURLRequest → AuditReportRelevance audit. 2–3 competitor_urls switch to relative benchmarking. profile picks a weighting, for example ai-focused.
      ApplyAuditActionsApplyAuditActionsRequest → ApplyAuditActionsResponsemode is plan (default), apply, commit, or pr. All but plan need PULSE_APPLY_ENABLED=true.

  string website_url = 8;
  repeated string goals = 9;
}

      The Diagnostic message

      channel_prescriptions and brand_visibility hold only proposals the judging ladder kept. judgements lists every verdict.

      message Diagnostic {
  string id = 1;
  string user_id = 2;
  string business_name = 3;
  string industry = 4;
  string niche = 5;
  string target_audience = 6;
  string current_channels = 7;      // comma-separated active channels
  string competitors = 8;           // comma-separated competitor names
  string business_description = 9;  // user's free-form business description
  string website_url = 10;
  string positioning_analysis = 11;
  string swot_analysis = 12;
  string channel_recommendations = 13;
  string content_strategy = 14;
  string competitive_landscape = 15;
  string ai_opinion = 16;           // strategic opinion from the AI market-analyst
  string status = 17;               // "pending", "generating", "completed", "error"
  map<string, string> metadata = 18;
  google.protobuf.Timestamp created_at = 19;
  google.protobuf.Timestamp updated_at = 20;
  repeated ChannelPrescription channel_prescriptions = 21;
  repeated AudienceSegment audience_segments = 22;
  repeated CronogramTask cronogram = 23;
  repeated string sources_used = 24;
  repeated string agent_tasks = 25;
  repeated string goals = 26;
  repeated DocumentPrescription document_prescriptions = 27;
  repeated BrandVisibilityInsight brand_visibility = 28;
  runink.ui.judgement.v1.JudgementSet judgements = 29;
}

message ChannelPrescription {
  Channel channel = 1;
  string role = 2;
  string priority = 3;
  repeated ContentSuggestion suggestions = 4;
  string cta_recommendation = 5;
  repeated string kpis = 6;
  runink.ui.judgement.v1.Judgement judgement = 7;
}

message ContentSuggestion {
  string title = 1;
  string description = 2;
  string cta = 3;
}

message DocumentPrescription {
  string format = 1;
  string role = 2;
  string priority = 3;
  repeated ContentSuggestion suggestions = 4;
}

message BrandVisibilityInsight {
  string dimension = 1;
  string insight = 2;               // what the diagnosis found about current AI-search visibility
  repeated string target_terms = 3; // entities/phrases/queries to own for AI-search lexical relevance
  repeated string actions = 4;      // concrete actions to raise generative-AI discoverability
  string rationale = 5;             // why this shapes brand awareness in AI/metasearch engines
  runink.ui.judgement.v1.Judgement judgement = 6;
}

message AudienceSegment {
  string level = 1;
  repeated string areas = 2;
  repeated string roles = 3;
  repeated string technologies = 4;

message CronogramTask {

  bool completed = 4;
}

message AuditAction {

  string confidence = 8;   // high|medium|low
  bool auto_apply = 9;

      RadarService

      Market research. See Audience and market.

      MethodRequest → ResponseNotes
      StartResearchStartResearchRequest → stream StreamChunkLive research for a topic, with optional extra sources
      GetRadarResultsGetRadarResultsRequest → RadarResultsOne stored result
      ListRadarResultsListRadarResultsRequest → ListRadarResultsResponseMost recent first
      SuggestProfilesSuggestProfilesRequest → SuggestProfilesResponseInfluencer, competitor, and partner profiles
      UseRadarFeedbackUseRadarFeedbackRequest → EmptyProfile-level feedback
      GetTrendingPostsGetTrendingPostsRequest → GetTrendingPostsResponseReal scraped posts with a 0–100 hot score
      SubmitPostFeedbackSubmitPostFeedbackRequest → Emptylike or dislike on one post
      DiscoverCompetitorChannelsDiscoverCompetitorChannelsRequest → DiscoverCompetitorChannelsResponseCompetitors’ YouTube and Instagram channels
      DiscoverGuestOpportunitiesDiscoverGuestOpportunitiesRequest → DiscoverGuestOpportunitiesResponseShows you could guest on
      DraftPartnershipOutreachDraftPartnershipOutreachRequest → stream StreamChunkDraft only. Nothing is sent.
      ListPartnershipOutreachListPartnershipOutreachRequest → ListPartnershipOutreachResponse
      message StartResearchRequest {
  string topic = 1;
  string niche = 2;
  repeated Channel channels = 3;
  string target_audience = 4;
  string region = 5;            // e.g. "BR", "US", "LATAM"
  string language = 6;          // e.g. "pt-BR", "en-US"
  repeated ExtraSource extra_sources = 7;
}

enum ExtraSourceKind {
  EXTRA_SOURCE_KIND_UNSPECIFIED = 0;
  EXTRA_SOURCE_KIND_WEB = 1;         // a web-search seed (query grounded live)
  EXTRA_SOURCE_KIND_WEBSITE = 2;     // a website or YouTube URL to scrape
  EXTRA_SOURCE_KIND_PASTED_TEXT = 3; // free-form text pasted by the user
}

message ExtraSource {
  ExtraSourceKind kind = 1;
  string label = 2;     // display title (page title, "Pasted text", the query…)
  string url = 3;       // for website/YouTube sources
  string content = 4;   // inline text for pasted-text sources
}

      InsightsService

      Content intelligence, synthesized by the platform’s own model from your diagnosis, published content, and live market signals. Every row passes through the judging ladder.

      MethodRequest → ResponseNotes
      AnalyzeContentGapsAnalyzeContentGapsRequest → ContentGapsResponseTopics and formats you under-serve
      GenerateContentStrategyGenerateContentStrategyRequest → ContentStrategyResponsePillars, cadence, alignment score
      GenerateStrategyRoadmapGenerateStrategyRoadmapRequest → StrategyRoadmapResponsePhased roadmap, readiness score
      message AnalyzeContentGapsRequest {
  string focus_area = 1;    // optional theme to concentrate the analysis on
  DateRange date_range = 2; // window of published content to assess
}

message ContentGap {
  string topic = 1;
  string format = 2;         // e.g. "article", "video", "infographic"
  double demand_score = 3;   // 0-100 — how much the audience wants this
  double coverage_score = 4; // 0-100 — how well the user currently covers it
  double opportunity = 5;    // 0-100 — prescriptive priority (demand vs coverage)
  string rationale = 6;
  runink.ui.judgement.v1.Judgement judgement = 7;
}

message ContentGapsResponse {
  repeated ContentGap gaps = 1;
  double coverage_index = 2; // headline: overall content coverage 0-100
  string summary = 3;
  google.protobuf.Timestamp generated_at = 4;
  runink.ui.judgement.v1.JudgementSet judgements = 5;
}

message GenerateContentStrategyRequest {
  string objective = 1;         // e.g. "Thought Leadership", "Lead Gen"
  repeated Channel channels = 2;
  string audience = 3;
}

message ContentPillar {
  string name = 1;
  string description = 2;
  repeated string example_topics = 3;
  double weight = 4; // recommended share of the content mix 0-100
  runink.ui.judgement.v1.Judgement judgement = 5;
}

message ContentStrategyResponse {
  repeated ContentPillar pillars = 1;
  string cadence = 2;          // e.g. "3 posts/week, 1 long-form/month"
  double alignment_score = 3;  // headline: strategy-fit 0-100
  string summary = 4;
  google.protobuf.Timestamp generated_at = 5;
  runink.ui.judgement.v1.JudgementSet judgements = 6;
}

message GenerateStrategyRoadmapRequest {
  string objective = 1;
  int32 horizon_weeks = 2; // planning horizon (defaults sensibly when 0)
}

message RoadmapPhase {
  string name = 1;                // e.g. "Foundation", "Momentum", "Scale"
  string timeframe = 2;           // e.g. "Weeks 1-4"
  repeated string initiatives = 3;
  double momentum = 4;            // projected momentum 0-100
  string outcome = 5;
  runink.ui.judgement.v1.Judgement judgement = 6;
}

message StrategyRoadmapResponse {
  repeated RoadmapPhase phases = 1;
  double readiness_score = 2; // headline 0-100
  string summary = 3;
  google.protobuf.Timestamp generated_at = 4;
  runink.ui.judgement.v1.JudgementSet judgements = 5;
}
